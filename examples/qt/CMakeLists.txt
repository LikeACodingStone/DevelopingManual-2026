cmake_minimum_required(VERSION 3.16)
project(handbook_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network Sql)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network Sql)
foreach(name json threads sqlite)
    add_executable(${name}_demo ${name}_demo.cpp)
    target_link_libraries(${name}_demo PRIVATE Qt${QT_VERSION_MAJOR}::Core)
endforeach()
target_link_libraries(sqlite_demo PRIVATE Qt${QT_VERSION_MAJOR}::Sql)
add_executable(udp_demo udp_demo.cpp)
target_link_libraries(udp_demo PRIVATE Qt${QT_VERSION_MAJOR}::Network)
